def find_agent_by_phone(phone):
    """Incoming call pe caller identify karo"""
    clean = phone.replace('+91', '').replace('+', '').strip()
    for agent in cached_contacts:
        agent_clean = agent['phone'].replace('+91', '').strip()
        if clean == agent_clean or clean[-10:] == agent_clean[-10:]:
            return agent
    return None

    return cached_contacts

@app.route('/incoming-call', methods=['POST'])
def incoming_call():
    """
    Incoming call handle karo
    Jab koi agent Twilio number pe call kare
    """
    caller = request.form.get('From', '')
    call_sid = request.form.get('CallSid', '')
    
    # Agent identify karo
    agent = find_agent_by_phone(caller)
    agent_name = agent['name'] if agent else 'Agent'
    
    print(f"📞 Incoming call from: {caller} — {agent_name}")
    
    # AI response generate karo
    response = VoiceResponse()
    
    # Hindi mein greet karo
    gather = Gather(
        input='speech',
        language='hi-IN',
        speech_timeout='auto',
        action=f'/handle-speech?agent={agent_name}&caller={caller}',
        method='POST'
    )
    
    gather.say(
        f"Namaste {agent_name} ji! Main aapka AI assistant hoon. "
        f"PB Partners mein aapki kaise madad kar sakta hoon? "
        f"Aap motor insurance, bike insurance, ya health insurance ke baare mein puchh sakte hain.",
        voice='Polly.Aditi',
        language='hi-IN'
    )
    
    response.append(gather)
    response.say("Koi response nahi mila. Phir call karein.", 
                 voice='Polly.Aditi', language='hi-IN')
    
    return Response(str(response), mimetype='text/xml')

@app.route('/handle-speech', methods=['POST'])
def handle_speech():
    """Agent ki baat sunke AI response do"""
    speech_result = request.form.get('SpeechResult', '')
    agent_name    = request.args.get('agent', 'Agent')
    caller        = request.args.get('caller', '')
    call_sid      = request.form.get('CallSid', '')
    
    print(f"🎤 Speech from {agent_name}: {speech_result}")
    
    if not speech_result:
        response = VoiceResponse()
        response.say("Maafi chahta hoon, mujhe samajh nahi aaya. Phir bolein.",
                    voice='Polly.Aditi', language='hi-IN')
        return Response(str(response), mimetype='text/xml')
    
    # OpenAI se AI response lo
    ai_reply = get_ai_response(speech_result, agent_name, call_sid)
    
    response = VoiceResponse()
    
    gather = Gather(
        input='speech',
        language='hi-IN', 
        speech_timeout='auto',
        action=f'/handle-speech?agent={agent_name}&caller={caller}',
        method='POST'
    )
    gather.say(ai_reply, voice='Polly.Aditi', language='hi-IN')
    response.append(gather)
    
    return Response(str(response), mimetype='text/xml')

@app.route('/make-call', methods=['POST'])
def make_call():
    """
    Outgoing call karo kisi agent ko
    POST: {"phone": "+919876543210", "message": "Reminder message"}
    """
    data    = request.json or {}
    phone   = data.get('phone', '')
    message = data.get('message', 'Namaste! Yeh ek reminder call hai PB Partners ki taraf se.')
    
    if not phone:
        return jsonify({'error': 'Phone number required'}), 400
    
    try:
        call = twilio_client.calls.create(
            to=phone,
            from_=TWILIO_FROM_NUMBER,
            twiml=f'''<Response>
                <Say voice="Polly.Aditi" language="hi-IN">{message}</Say>
            </Response>'''
        )
        print(f"📲 Outgoing call to {phone}: {call.sid}")
        return jsonify({'success': True, 'call_sid': call.sid, 'to': phone})
    except Exception as e:
        print(f"❌ Call error: {e}")
        return jsonify({'error': str(e)}), 500

def get_ai_response(user_message, agent_name, call_sid):
    """OpenAI se insurance guide response lo"""
    try:
        # Conversation history
        if call_sid not in conversations:
            conversations[call_sid] = []
        
        conversations[call_sid].append({
            "role": "user", 
            "content": user_message
        })
        
        system_prompt = f"""Tu ek expert insurance AI assistant hai — Agent Guru.
Tu {agent_name} ji se baat kar raha hai jo ek PB Partners insurance agent hai.

Tu inhe guide karta hai:
- Motor/Car insurance quote karne mein
- Bike insurance karne mein  
- Health insurance karne mein
- Third Party policy mein
- Customer se kaise baat karein

Rules:
- Hamesha Hindi mein baat kar
- Simple aur clear language use kar
- Step by step guide kar
- Encouragement de
- 2-3 sentences mein jawab de (call pe zyada lamba nahi)
- Professional lekin friendly reh"""

        response = openai.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": system_prompt},
                *conversations[call_sid][-6:]  # Last 6 messages
            ],
            max_tokens=150,
            temperature=0.7
        )
        
        ai_reply = response.choices[0].message.content
        conversations[call_sid].append({"role": "assistant", "content": ai_reply})
        
        print(f"🤖 AI: {ai_reply}")
        return ai_reply
        
    except Exception as e:
